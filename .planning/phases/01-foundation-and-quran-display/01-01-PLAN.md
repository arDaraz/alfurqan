    export function toArabicIndic(num: number): string {
      return num.toString().replace(/\d/g, (d) => String.fromCharCode(0x0660 + parseInt(d)));

    /** Normalize Arabic text for search (strip diacritics) */

    CREATE TABLE surahs (
