    1. **Selection state management:**
       - Import `useSelectionStore` from stores.
       - Derive each ayah's selectionState:
         ```typescript
         function getSelectionState(ayahNumber: number): 'default' | 'selected-start' | 'selected-end' | 'in-range' {
           if (startAyah === ayahNumber && startSurah === surahNumber) return 'selected-start';
           if (endAyah === ayahNumber && endSurah === surahNumber) return 'selected-end';
           if (startAyah && endAyah && startSurah === surahNumber && ayahNumber > startAyah && ayahNumber < endAyah) return 'in-range';
           return 'default';
         }
         ```
       - Handle ayah tap:
         ```typescript
         function handleAyahPress(ayahNumber: number) {
           if (!startAyah || (startAyah && endAyah)) {
             // Start new selection (or restart after complete selection)
             clearSelection();
             setStart(surahNumber, ayahNumber);
           } else {
             // Complete the range
             if (ayahNumber < startAyah) {
               // Tapped before start — swap
               setEnd(surahNumber, startAyah);
               setStart(surahNumber, ayahNumber);
             } else if (ayahNumber === startAyah) {
               // Tapped same ayah — deselect
               clearSelection();
             } else {
               setEnd(surahNumber, ayahNumber);
             }
           }
         }
         ```
